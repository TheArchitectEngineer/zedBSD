[pi5]
kernel=vmunix
# vmunix is linked to run at 0x80000, just above the secure firmware (BL31)
# the bootloader keeps at 0..0x80000.
kernel_address=0x80000
# No device_tree line: the bootloader picks bcm2712-rpi-5-b.dtb (C1) or
# bcm2712d0-rpi-5-b.dtb (D0) by the chip's stepping, and both are here.
# The firmware framebuffer the console draws on: 32 bits per pixel, a
# 1280x720 surface scaled to the display's mode, no overscan border.
framebuffer_depth=32
framebuffer_ignore_alpha=1
framebuffer_width=1280
framebuffer_height=720
disable_overscan=1
[all]
